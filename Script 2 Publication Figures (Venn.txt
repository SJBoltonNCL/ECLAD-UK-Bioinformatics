# --- FIGURE 2: Global Heatmap with Margin Fixes ---
log_cpms <- cpm(dge, log = TRUE)
full_heatmap_matrix <- log_cpms[unique(c(w12_names, w24_names)), ]

png("Figure2_Global_Heatmap.png", width = 3000, height = 2400, res = 300)
heatmap.2(full_heatmap_matrix, trace = "none", 
          col = colorRampPalette(c("#002060", "white", "#C00000"))(100), 
          scale = "row", density.info = "none",
          margin = c(12, 10), cexRow = 0.45, cexCol = 0.65, srtCol = 45, adjCol = c(1, 1),
          lhei = c(1.5, 8), keysize = 1.0,
          main = "Global Longitudinal Profile (87 Active Candidate Genes)")
dev.off()
